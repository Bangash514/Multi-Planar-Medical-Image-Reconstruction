cmake_minimum_required(VERSION 3.12)
project(MultiPlanarReconstruction C CXX)
set(CMAKE_CXX_STANDARD 11)

# VTK 9 version (axial, coronal, sagittal, oblique)
find_package(VTK COMPONENTS CommonCore CommonDataModel CommonMath CommonTransforms ImagingCore IOImage)
if(VTK_FOUND)
  add_executable(mpr_slicer mpr_slicer.cpp)
  target_link_libraries(mpr_slicer PRIVATE ${VTK_LIBRARIES})
  vtk_module_autoinit(TARGETS mpr_slicer MODULES ${VTK_LIBRARIES})
endif()

# ITK 5 version (axial, coronal, sagittal)
find_package(ITK)
if(ITK_FOUND)
  include(${ITK_USE_FILE})
  add_executable(mpr_itk mpr_itk.cpp)
  target_link_libraries(mpr_itk ${ITK_LIBRARIES})
endif()
